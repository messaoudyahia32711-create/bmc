     1→// ─── PPTX pitch deck v2 (client-side, pptxgenjs): 14 data-rich slides ────────
     2→import { Project, Lang, Branding } from './types';
     3→import { computeSnapshot, getStartupBudget, buildScfYear } from './engine';
     4→import { getDict } from './i18n';
     5→import { getX } from './i18n2';
     6→import { money as fmtMoney } from './format';
     7→import { getDoc, DocKey } from './annex-docs';
     8→import { matchPrograms } from './knowledge';
     9→import { completeness } from './analytics';
    10→
    11→export async function generatePptx(p: Project, lang: Lang, branding: Branding): Promise<void> {
    12→  const { default: PptxGenJS } = await import('pptxgenjs');
    13→  const t = getDict(lang);
    14→  const x = getX(lang);
    15→  const money = (n: number) => fmtMoney(n, lang);
    16→  const md = (n: number) => n >= 1e6 ? `${(n / 1e6).toLocaleString('fr-DZ', { maximumFractionDigits: 1 })} ${x.dMd}` : money(n);
    17→  const snap = computeSnapshot(p);
    18→  const budget = getStartupBudget(p);
    19→  const need = snap.assets.invest + budget.workingCapital;
    20→  const comp = completeness(p);
    21→  const rtl = lang === 'ar';
    22→  const orgName = branding.orgName || t.appName;
    23→  const primary = (branding.primary || '#2E5E73').replace('#', '');
    24→  const primaryDark = '1E3D4C';
    25→  const gold = 'C99A3C', green = '3E7C59', red = 'AC4F46', amber = '96691F';
    26→  const ink = '22313A', sub = '4A5B68', faint = '8FA0AD', line = 'E2E9EE', panel = 'F7FAFB';
    27→  const font = 'Cairo';
    28→  const align = rtl ? ('right' as const) : ('left' as const);
    29→  /** Wrap a value in LEFT-TO-RIGHT MARKs so digit groups (phone numbers) keep
    30→   *  their natural order inside Arabic RTL text boxes in PowerPoint. */
    31→  const lrm = (s: string) => (s ? `\u200e${s}\u200e` : s);
    32→  const projectName = p.info.projectTitleAr || p.name || t.untitled;
    33→  const legalLabel = { micro: t.legalMicro, startup: t.legalStartup, patent: t.legalPatent, other: t.legalOther }[p.info.legal];
    34→  const lines = (v?: string): string[] => (v || '').split('\n').map(s => s.trim().replace(/^•\s*/, '')).filter(Boolean);
    35→  const programs = matchPrograms(p).filter(pr => pr.fit !== 'low').slice(0, 3).map(pr => (t as unknown as Record<string, string>)[pr.nameKey]);
    36→
    37→  const pres = new PptxGenJS();
    38→  pres.layout = 'LAYOUT_16x9';
    39→  pres.rtlMode = rtl;
    40→  pres.theme = { headFontFace: font, bodyFontFace: font };
    41→  pres.title = projectName;
    42→
    43→  const W = 10, H = 5.625;
    44→  const TOTAL = 14;
    45→
    46→  const txt = (s: PptxGenJS.Slide, text: string, opts: Partial<PptxGenJS.TextProps> = {}) => {
    47→    s.addText(text, { fontFace: font, align, rtlMode: rtl, color: ink, ...opts } as PptxGenJS.TextProps);
    48→  };
    49→  const bulletOpts = { fontFace: font, align, rtlMode: rtl, valign: 'top' as const };
    50→  const bullets = (items: string[]) => items.filter(Boolean).map(l => ({ text: l, options: { bullet: true, breakLine: true } }));
    51→
    52→  const baseSlide = (n: number) => {
    53→    const s = pres.addSlide();
    54→    s.background = { color: 'FFFFFF' };
    55→    s.addShape('rect', { x: 0, y: 0, w: W, h: 0.12, fill: { color: primary } });
    56→    s.addShape('rect', { x: rtl ? 0.4 : W - 1.2, y: 0.62, w: 0.8, h: 0.05, fill: { color: gold } });
    57→    s.addText(`${orgName}`, { x: 0.5, y: H - 0.34, w: W / 2, h: 0.3, fontSize: 8.5, color: faint, fontFace: font, align: rtl ? 'right' : 'left', rtlMode: rtl });
    58→    s.addText(`${t.slideOf} ${n} / ${TOTAL}`, { x: W / 2, y: H - 0.34, w: W / 2 - 0.5, h: 0.3, fontSize: 8.5, color: faint, fontFace: font, align: rtl ? 'left' : 'right', rtlMode: rtl });
    59→    return s;
    60→  };
    61→  const addTitle = (s: PptxGenJS.Slide, title: string) => {
    62→    txt(s, title, { x: 0.5, y: 0.28, w: W - 1, h: 0.55, fontSize: 23, bold: true, color: primary });
    63→  };
    64→  const kpiCard = (s: PptxGenJS.Slide, i: number, count: number, row: number, k: string, v: string, color = primary, y0?: number, h0?: number) => {
    65→    const cw = (W - 1.2) / count, x0 = 0.6 + i * cw + 0.06, yy = y0 ?? (row === 0 ? 1.15 : 2.75), hh = h0 ?? 1.45;
    66→    s.addShape('roundRect', { x: x0, y: yy, w: cw - 0.12, h: hh, fill: { color: panel }, line: { color: line }, rectRadius: 0.08 });
    67→    s.addText(k, { x: x0 + 0.1, y: yy + 0.1, w: cw - 0.32, h: 0.3, fontSize: 9.5, color: faint, fontFace: font, align, rtlMode: rtl });
    68→    s.addText(v, { x: x0 + 0.1, y: yy + 0.42, w: cw - 0.32, h: hh - 0.5, fontSize: 17, bold: true, color, fontFace: font, align, rtlMode: rtl });
    69→  };
    70→  /** Reference note (definition + formulas) — same static content as the report. */
    71→  const addRefNote = (s: PptxGenJS.Slide, k: DocKey, y: number, boxed = false) => {
    72→    const d = getDoc(lang, k);
    73→    const fexpr = d.formulas.slice(0, 2).map(f => f.expr).join('      ');
    74→    if (boxed) {
    75→      s.addShape('roundRect', { x: 0.6, y, w: W - 1.2, h: 0.85, fill: { color: 'F7FAFB' }, line: { color: 'C9DAE5', dashType: 'dash' }, rectRadius: 0.05 });
    76→      s.addText([
    77→        { text: `${x.dDocLabel}: ${d.short}`, options: { fontFace: font, color: sub, breakLine: true } },
    78→        { text: fexpr, options: { fontFace: 'Consolas', color: primary } },
    79→      ], { x: 0.75, y: y + 0.04, w: W - 1.5, h: 0.78, fontSize: 9, align: rtl ? 'right' : 'left', valign: 'middle', rtlMode: rtl });
    80→    } else {
    81→      s.addText([
    82→        { text: `${x.dDocLabel}: ${d.short}`, options: { fontFace: font, color: faint } },
    83→        { text: `   ${fexpr}`, options: { fontFace: 'Consolas', color: faint } },
    84→      ], { x: 0.6, y, w: W - 1.2, h: 0.3, fontSize: 8.5, align: rtl ? 'right' : 'left', valign: 'middle', rtlMode: rtl });
    85→    }
    86→  };
    87→  const tableHead = (labels: string[]) => labels.map(l => ({ text: l, options: { bold: true, color: 'FFFFFF', fill: { color: primary }, fontFace: font } }));
    88→
    89→  // ── 1 · Cover ──
    90→  {
    91→    const s = pres.addSlide();
    92→    s.background = { color: primary };
    93→    s.addShape('ellipse', { x: W - 2.2, y: -1, w: 3.4, h: 3.4, fill: { color: 'FFFFFF', transparency: 94 } });
    94→    s.addShape('ellipse', { x: -1.2, y: H - 2.2, w: 3.6, h: 3.6, fill: { color: 'FFFFFF', transparency: 94 } });
    95→    if (branding.logo) { try { s.addImage({ data: branding.logo, x: W / 2 - 0.55, y: 0.42, w: 1.1, h: 1.1 }); } catch { /* ignore */ } }
    96→    s.addShape('roundRect', { x: W / 2 - 1.55, y: branding.logo ? 1.72 : 1.5, w: 3.1, h: 0.42, fill: { color: 'FFFFFF', transparency: 88 }, line: { color: 'FFFFFF', transparency: 60 }, rectRadius: 0.21 });
    97→    s.addText(x.dBadge, { x: W / 2 - 1.55, y: branding.logo ? 1.72 : 1.5, w: 3.1, h: 0.42, fontSize: 11, bold: true, color: 'FFFFFF', align: 'center', fontFace: font });
    98→    s.addText(projectName, { x: 0.5, y: 2.3, w: W - 1, h: 0.95, fontSize: 36, bold: true, color: 'FFFFFF', align: 'center', fontFace: font });
    99→    if (p.info.projectTitleEn) s.addText(p.info.projectTitleEn, { x: 0.5, y: 3.2, w: W - 1, h: 0.4, fontSize: 15, color: 'D9E7EE', align: 'center', fontFace: font });
   100→    const meta = [p.info.ownerName, p.info.wilaya, p.info.incubator || orgName].filter(Boolean).join('   ·   ');
   101→    s.addText(meta, { x: 0.5, y: 3.75, w: W - 1, h: 0.4, fontSize: 12.5, color: 'EAF2F6', align: 'center', fontFace: font });
   102→    s.addShape('roundRect', { x: W / 2 - 1.25, y: 4.35, w: 2.5, h: 0.62, fill: { color: 'FFFFFF', transparency: 86 }, line: { color: 'FFFFFF', transparency: 55 }, rectRadius: 0.1 });
   103→    s.addText([
   104→      { text: `${x.dScore}  `, options: { fontSize: 10.5, color: 'D9E7EE' } },
   105→      { text: snap.score.toFixed(1), options: { fontSize: 20, bold: true, color: 'FFFFFF' } },
   106→      { text: ' / 10', options: { fontSize: 11, color: 'D9E7EE' } },
   107→    ], { x: W / 2 - 1.25, y: 4.35, w: 2.5, h: 0.62, align: 'center', valign: 'middle', fontFace: font });
   108→    s.addText(x.dToolCredits, { x: 0.5, y: 5.14, w: W - 1, h: 0.3, fontSize: 9, color: 'BFD5E0', align: 'center', fontFace: font });
   109→  }
   110→
   111→  // ── 2 · Project card ──
   112→  {
   113→    const s = baseSlide(2);
   114→    addTitle(s, `${x.dCardT} — ${x.dCardD}`);
   115→    const tiles: [string, string][] = [
   116→      [t.tradeName, p.info.tradeName || t.na],
   117→      [x.dLegal, legalLabel],
   118→      [t.ownerName, p.info.ownerName || t.na],
   119→      [t.wilaya, p.info.wilaya || t.na],
   120→      [t.phone, p.info.phone ? lrm(p.info.phone) : t.na],
   121→      [x.dDate, p.info.preparedDate || new Date().toLocaleDateString('fr-DZ')],
   122→    ];
   123→    tiles.forEach(([k, v], i) => {
   124→      const cw = (W - 1.2) / 3;
   125→      const cx = 0.6 + (i % 3) * cw + 0.05, cy = 1.05 + Math.floor(i / 3) * 0.95;
   126→      s.addShape('roundRect', { x: cx, y: cy, w: cw - 0.1, h: 0.82, fill: { color: panel }, line: { color: line }, rectRadius: 0.06 });
   127→      s.addText(k, { x: cx + 0.1, y: cy + 0.08, w: cw - 0.2, h: 0.25, fontSize: 8.5, color: faint, fontFace: font, align, rtlMode: rtl });
   128→      s.addText(v, { x: cx + 0.1, y: cy + 0.34, w: cw - 0.2, h: 0.42, fontSize: 12.5, bold: true, color: ink, fontFace: font, align, rtlMode: rtl });
   129→    });
   130→    txt(s, p.info.description || t.na, { x: 0.6, y: 3.05, w: W - 1.2, h: 1.15, fontSize: 12.5, color: sub, valign: 'top' });
   131→    s.addShape('roundRect', { x: 0.6, y: 4.3, w: W - 1.2, h: 0.55, fill: { color: 'FFFFFF' }, line: { color: line }, rectRadius: 0.06 });
   132→    s.addText(`${x.dCompletion}`, { x: 0.75, y: 4.3, w: 1.7, h: 0.55, fontSize: 10, color: faint, fontFace: font, align, rtlMode: rtl, valign: 'middle' });
   133→    s.addShape('roundRect', { x: 2.5, y: 4.48, w: 4.6, h: 0.2, fill: { color: 'EAF0F3' }, rectRadius: 0.1 });
   134→    s.addShape('roundRect', { x: 2.5, y: 4.48, w: Math.max(0.2, 4.6 * comp.percent / 100), h: 0.2, fill: { color: comp.percent >= 80 ? green : gold }, rectRadius: 0.1 });
   135→    s.addText(`${comp.percent}%`, { x: 7.2, y: 4.3, w: 1.4, h: 0.55, fontSize: 12, bold: true, color: primary, fontFace: font, align: rtl ? 'left' : 'right', valign: 'middle' });
   136→  }
   137→
   138→  // ── 3 · Problem ──
   139→  {
   140→    const s = baseSlide(3);
   141→    addTitle(s, `${t.dProblem} — ${t.dProblemT}`);
   142→    txt(s, p.info.description || t.na, { x: 0.6, y: 1.1, w: W / 2 - 0.1, h: 1.7, fontSize: 13, color: ink, valign: 'top' });
   143→    s.addText(bullets(lines(p.market.customers)) as never, { x: 0.6, y: 2.9, w: W / 2 - 0.1, h: 2.2, fontSize: 12, color: sub, ...bulletOpts });
   144→    const segs = lines(p.market.customers).slice(0, 3);
   145→    segs.forEach((c, i) => {
   146→      s.addShape('roundRect', { x: W / 2 + 0.15, y: 1.15 + i * 1.15, w: W / 2 - 0.85, h: 1.0, fill: { color: panel }, line: { color: line }, rectRadius: 0.07 });
   147→      s.addText(c, { x: W / 2 + 0.3, y: 1.15 + i * 1.15, w: W / 2 - 1.15, h: 1.0, fontSize: 11.5, bold: true, color: sub, fontFace: font, align, rtlMode: rtl, valign: 'middle' });
   148→    });
   149→  }
   150→
   151→  // ── 4 · Solution ──
   152→  {
   153→    const s = baseSlide(4);
   154→    addTitle(s, `${t.dSolution} — ${t.dSolutionT}`);
   155→    lines(p.market.valueProp).slice(0, 4).forEach((v, i) => {
   156→      const y0 = 1.15 + i * 0.85;
   157→      s.addShape('roundRect', { x: 0.6, y: y0, w: W - 1.2, h: 0.72, fill: { color: 'E7F2EB' }, line: { color: 'C6DFD0' }, rectRadius: 0.07 });
   158→      s.addText(`✓  ${v}`, { x: 0.8, y: y0, w: W - 1.6, h: 0.72, fontSize: 12.5, bold: true, color: '2A5646', fontFace: font, align, rtlMode: rtl, valign: 'middle' });
   159→    });
   160→    txt(s, `${t.bmcBlocks[1]}: ${lines(p.market.activities).join(' · ') || t.na}`, { x: 0.6, y: 4.75, w: W - 1.2, h: 0.6, fontSize: 10.5, color: sub, valign: 'top' });
   161→  }
   162→
   163→  // ── 5 · Market ──
   164→  {
   165→    const s = baseSlide(5);
   166→    addTitle(s, `${t.dMarket} — ${t.dMarketT}`);
   167→    s.addText(bullets(lines(p.market.customers).slice(0, 4)) as never, { x: 0.6, y: 1.15, w: W / 2 - 0.2, h: 2.3, fontSize: 12, color: ink, ...bulletOpts });
   168→    const totalBuyers = p.market.personas.reduce((a, pe) => a + (pe.monthlyBuyers || 0), 0);
   169→    const marginRate = snap.base.rev > 0 ? (snap.base.rev - snap.base.varCost) / snap.base.rev : 0;
   170→    s.addShape('roundRect', { x: 0.6, y: 3.6, w: W / 2 - 0.35, h: 1.45, fill: { color: primary, transparency: 92 }, line: { color: primary, transparency: 70 }, rectRadius: 0.08 });
   171→    s.addText(x.dBuyers, { x: 0.7, y: 3.7, w: W / 2 - 0.55, h: 0.3, fontSize: 10, color: faint, fontFace: font, align: 'center' });
   172→    s.addText(totalBuyers.toLocaleString('fr-DZ'), { x: 0.7, y: 3.98, w: W / 2 - 0.55, h: 0.7, fontSize: 26, bold: true, color: primary, fontFace: font, align: 'center' });
   173→    s.addText(`${x.dMargin}: ${(marginRate * 100).toFixed(0)}%`, { x: 0.7, y: 4.66, w: W / 2 - 0.55, h: 0.3, fontSize: 10, color: faint, fontFace: font, align: 'center' });
   174→    if (p.market.personas.length) {
   175→      const rows = p.market.personas.slice(0, 5).map(pe => [
   176→        { text: pe.name || '—', options: { bold: true, fontFace: font } },
   177→        { text: pe.ageRange || '—', options: { fontFace: font } },
   178→        { text: pe.location || '—', options: { fontFace: font } },
   179→        { text: (pe.monthlyBuyers || 0).toLocaleString('fr-DZ'), options: { fontFace: font } },
   180→      ]);
   181→      s.addTable([
   182→        tableHead([t.peName, t.peAge, t.peLocation, t.peMonthly]),
   183→        ...rows,
   184→      ] as never, { x: W / 2 + 0.1, y: 1.2, w: W / 2 - 0.7, fontSize: 10.5, align: rtl ? 'right' : 'left', border: { pt: 0.5, color: line }, valign: 'middle' });
   185→    }
   186→  }
   187→
   188→  // ── 6 · Competitors & SWOT ──
   189→  {
   190→    const s = baseSlide(6);
   191→    addTitle(s, `${x.dCompSwot} — ${x.dCompSwotT}`);
   192→    if (p.market.competitors.length) {
   193→      const rows = p.market.competitors.slice(0, 3).map(cp => [
   194→        { text: cp.name || '—', options: { bold: true, fontFace: font } },
   195→        { text: cp.offer || '—', options: { fontFace: font } },
   196→        { text: cp.edge || '—', options: { fontFace: font } },
   197→      ]);
   198→      s.addTable([tableHead([t.compName, t.compOffer, t.compEdge]), ...rows] as never,
   199→        { x: 0.6, y: 1.05, w: W - 1.2, fontSize: 10, align: rtl ? 'right' : 'left', border: { pt: 0.5, color: line } });
   200→    }
   201→    const q = (x0: number, y0: number, w: number, h: number, label: string, items: string[], head: string, bg: string, bd: string) => {
   202→      s.addShape('roundRect', { x: x0, y: y0, w, h, fill: { color: bg }, line: { color: bd }, rectRadius: 0.07 });
   203→      s.addText(label, { x: x0 + 0.12, y: y0 + 0.05, w: w - 0.24, h: 0.3, fontSize: 10.5, bold: true, color: head, fontFace: font, align, rtlMode: rtl });
   204→      s.addText(bullets(items.slice(0, 3)) as never, { x: x0 + 0.18, y: y0 + 0.36, w: w - 0.36, h: h - 0.44, fontSize: 9, color: sub, ...bulletOpts });
   205→    };
   206→    const qw = (W - 1.3) / 2, qh = 1.42;
   207→    q(0.6, 2.6, qw, qh, t.swotS, p.market.swot.strengths, green, 'E7F2EB', 'C6DFD0');
   208→    q(0.6 + qw + 0.1, 2.6, qw, qh, t.swotW, p.market.swot.weaknesses, red, 'F7E7E5', 'E8C7C3');
   209→    q(0.6, 2.6 + qh + 0.1, qw, qh, t.swotO, p.market.swot.opportunities, primary, 'EAF0F5', 'C9DAE5');
   210→    q(0.6 + qw + 0.1, 2.6 + qh + 0.1, qw, qh, t.swotT, p.market.swot.threats, amber, 'F6EFDD', 'E5D5AE');
   211→  }
   212→
   213→  // ── 7 · BMC ──
   214→  {
   215→    const s = baseSlide(7);
   216→    addTitle(s, `${t.dBmc} — ${t.dBmcT}`);
   217→    const cells: [string, string][] = [
   218→      [t.bmcBlocks[0], lines(p.market.partners).slice(0, 3).join(' · ') || t.na],
   219→      [t.bmcBlocks[1], lines(p.market.activities).slice(0, 3).join(' · ') || t.na],
   220→      [t.bmcBlocks[3], lines(p.market.valueProp).slice(0, 3).join(' · ') || t.na],
   221→      [t.bmcBlocks[5], lines(p.market.channels).slice(0, 3).join(' · ') || t.na],
   222→      [t.bmcBlocks[6], lines(p.market.customers).slice(0, 3).join(' · ') || t.na],
   223→      [t.bmcBlocks[7], `${money(snap.fixed)} / ${t.monthUnit}`],
   224→      [t.bmcBlocks[8], `${money(snap.base.rev)} / ${t.monthUnit}`],
   225→    ];
   226→    s.addTable(cells.map(([k, v]) => ([
   227→      { text: k, options: { bold: true, color: 'FFFFFF', fill: { color: primary }, fontFace: font } },
   228→      { text: v, options: { fontFace: font } },
   229→    ])) as never, { x: 0.6, y: 1.15, w: W - 1.2, fontSize: 10.5, border: { pt: 0.5, color: line }, align: rtl ? 'right' : 'left' });
   230→    txt(s, x.dBmcHint, { x: 0.6, y: 4.9, w: W - 1.2, h: 0.3, fontSize: 9, italic: true, color: faint });
   231→  }
   232→
   233→  // ── 8 · Revenue model ──
   234→  {
   235→    const s = baseSlide(8);
   236→    addTitle(s, `${t.dModel} — ${t.dModelT}`);
   237→    const head = tableHead([t.productName, t.revenueModel, t.pPrice, t.pQty, t.ueMonthlyRev, t.ueMonthlyMargin]);
   238→    const rows = p.products.filter(z => z.name).map(z => {
   239→      const rev = z.model === 'fixed' ? z.price * z.qty : z.model === 'commission' ? z.deal * (z.commission / 100) * z.deals : z.sub * z.subs;
   240→      const marg = z.model === 'fixed' ? (z.price - z.variable) * z.qty : z.model === 'commission' ? (z.deal * z.commission / 100 - z.dealCost) * z.deals : (z.sub - z.subCost) * z.subs;
   241→      return [
   242→        { text: z.name, options: { bold: true, fontFace: font } },
   243→        { text: { fixed: t.rmFixed, commission: t.rmCommission, subscription: t.rmSubscription }[z.model], options: { fontFace: font } },
   244→        { text: money(z.model === 'fixed' ? z.price : z.model === 'commission' ? z.deal * z.commission / 100 : z.sub), options: { fontFace: font } },
   245→        { text: Math.round(z.model === 'fixed' ? z.qty : z.model === 'commission' ? z.deals : z.subs).toLocaleString('fr-DZ'), options: { fontFace: font } },
   246→        { text: money(rev), options: { fontFace: font } },
   247→        { text: money(marg), options: { fontFace: font, color: marg >= 0 ? green : red } },
   248→      ];
   249→    });
   250→    if (rows.length) {
   251→      s.addTable([head, ...rows, [
   252→        { text: t.total, options: { bold: true, color: 'FFFFFF', fill: { color: primaryDark }, fontFace: font, colSpan: 4 } },
